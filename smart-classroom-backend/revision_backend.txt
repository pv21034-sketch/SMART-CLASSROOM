impl Config {
    /// Construye y valida la configuraciÃ³n del backend.
    pub fn desde_entorno() -> Result<Self> {
        let host = leer_variable_compatible("SERVER_HOST", "HOST", "0.0.0.0")?;

        let puerto = leer_variable_compatible("SERVER_PORT", "PORT", "3000")?;

        // Validar direcciÃ³n IP.
        let ip: IpAddr = host.parse().with_context(|| {
            format!("SERVER_HOST/HOST debe ser una direcciÃ³n IP vÃ¡lida: {host}")
        })?;

        // Validar puerto.
        let puerto: u16 = puerto
            .parse()
            .with_context(|| "SERVER_PORT/PORT debe ser un nÃºmero entre 1 y 65535")?;

        if puerto == 0 {
            bail!("El puerto del servidor no puede ser 0");
        }

        let direccion = SocketAddr::new(ip, puerto);

        // JWT obligatorio.
        let jwt_secret =
            env::var("JWT_SECRET").context("Falta JWT_SECRET en las variables de entorno")?;

        if jwt_secret.trim().is_empty() {
            bail!("JWT_SECRET no puede estar vacÃ­o");
        }

        if jwt_secret.as_bytes().len() < 32 {
            bail!("JWT_SECRET debe contener al menos 32 bytes");
        }

        // Credenciales opcionales del administrador.
        let admin_usuario = leer_opcional("ADMIN_USUARIO")?;
        let admin_password = leer_opcional("ADMIN_PASSWORD")?;

        // Evitar configuraciones incompletas.
        if admin_usuario.is_some() != admin_password.is_some() {
            bail!("ADMIN_USUARIO y ADMIN_PASSWORD deben configurarse juntos");
        }

        if let Some(ref usuario) = admin_usuario {
            if usuario.chars().count() < 3 {
                bail!("ADMIN_USUARIO debe contener al menos 3 caracteres");
            }
        }

        if let Some(ref password) = admin_password {
            if password.chars().count() < 12 {
                bail!("ADMIN_PASSWORD debe contener al menos 12 caracteres");
            }
        }

        Ok(Self {
            direccion,
            jwt_secret,
            admin_usuario,
            admin_password,
        })
    }
}

/// Lee la variable principal o su alternativa.
///
/// Prioridad:
/// 1. Variable principal.
/// 2. Variable alternativa.
/// 3. Valor predeterminado.
fn leer_variable_compatible(
    principal: &str,
    alternativa: &str,
    predeterminado: &str,
) -> Result<String> {
    if let Some(valor) = leer_opcional(principal)? {
        return Ok(valor);
    }

    if let Some(valor) = leer_opcional(alternativa)? {
        return Ok(valor);
    }

    Ok(predeterminado.to_string())
}

/// Lee una variable opcional y descarta cadenas vacÃ­as.
fn leer_opcional(nombre: &str) -> Result<Option<String>> {
    match env::var(nombre) {
        Ok(valor) => {
            let valor = valor.trim();

            if valor.is_empty() {
                Ok(None)
            } else {
                Ok(Some(valor.to_string()))
            }
        }

        Err(env::VarError::NotPresent) => Ok(None),

        Err(error) => Err(error).with_context(|| format!("No se pudo leer la variable {nombre}")),
    }
}

/// Crea un pool de conexiones PostgreSQL.
///
/// Variables:
/// - DATABASE_URL
/// - DB_MAX_CONNECTIONS (por defecto 5)
/// - DB_MIN_CONNECTIONS (por defecto 0)
/// - DB_ACQUIRE_TIMEOUT (por defecto 10 segundos)
///
/// Los valores predeterminados son conservadores
/// para entornos con conexiones limitadas.
pub async fn crear_pool() -> Result<PgPool, sqlx::Error> {
    // Obtener URL.
    let database_url = env::var("DATABASE_URL")
        .map_err(|_| error_configuracion("DATABASE_URL no estÃ¡ configurada".to_string()))?;

    if database_url.trim().is_empty() {
        return Err(error_configuracion(
            "DATABASE_URL no puede estar vacÃ­a".to_string(),
        ));
    }

    // ConfiguraciÃ³n del pool.
    let max_conexiones = leer_entero("DB_MAX_CONNECTIONS", 5)?;

    let min_conexiones = match env::var("DB_MIN_CONNECTIONS") {
        Ok(valor) => valor.parse::<u32>().map_err(|_| {
            error_configuracion("DB_MIN_CONNECTIONS debe ser un entero no negativo".to_string())
        })?,

        Err(env::VarError::NotPresent) => 0,

        Err(error) => {
            return Err(error_configuracion(format!(
                "No se pudo leer DB_MIN_CONNECTIONS: {error}"
            )));
        }
    };

    if min_conexiones > max_conexiones {
        return Err(error_configuracion(
            "DB_MIN_CONNECTIONS no puede superar DB_MAX_CONNECTIONS".to_string(),
        ));
    }

    let timeout = leer_entero("DB_ACQUIRE_TIMEOUT", 10)?;

    tracing::info!(
        "Configurando pool PostgreSQL: mÃ­nimo={}, mÃ¡ximo={}",
        min_conexiones,
        max_conexiones
    );

    // Crear y conectar el pool.
    let pool = PgPoolOptions::new()
        .max_connections(max_conexiones)
        .min_connections(min_conexiones)
        .acquire_timeout(Duration::from_secs(u64::from(timeout)))
        .idle_timeout(Duration::from_secs(300))
        .max_lifetime(Duration::from_secs(1800))
        .connect(&database_url)
        .await?;

    tracing::info!("Pool de conexiones PostgreSQL creado correctamente");

    Ok(pool)
}

// Estructura que se envÃ­a al cliente en formato JSON cuando ocurre un error.
// AsÃ­ TODOS los errores de la API tienen la misma forma (consistentes).
#[derive(Serialize)]
pub struct ErrorResponse {
    pub status: u16,
    pub message: String,
}

// Enum central de errores. Cada variante representa un tipo de error distinto
// que puede ocurrir en cualquier parte de la aplicaciÃ³n (rutas, servicios, websocket, etc.)
// #[derive(Error)] (de la crate `thiserror`) genera automÃ¡ticamente el trait `std::error::Error`
// y el mensaje de cada variante se define con el atributo #[error("...")]
#[derive(Debug, Error)]
pub enum ApiError {
    #[error("Recurso no encontrado: {0}")]
    NotFound(String),

    #[error("Solicitud invÃ¡lida: {0}")]
    BadRequest(String),

    #[error("Acceso denegado: {0}")]
    Forbidden(String),

    #[error("Error interno del servidor: {0}")]
    Internal(String),

    #[error("Error de WebSocket: {0}")]
    WebSocket(String),
}

// IntoResponse es el trait de axum que le dice al framework cÃ³mo convertir
// un valor en una respuesta HTTP real. Al implementarlo para ApiError,
// podemos hacer `return Err(ApiError::NotFound(...))` directamente en un handler
// y axum lo transforma solo en la respuesta correcta.
impl IntoResponse for ApiError {
    fn into_response(self) -> Response {
        // Elegimos el cÃ³digo de estado HTTP segÃºn el tipo de error.
        // El `&self` es porque solo estamos "mirando" el valor, no consumiÃ©ndolo todavÃ­a.
        let status = match &self {
            ApiError::NotFound(_) => StatusCode::NOT_FOUND, // 404
            ApiError::BadRequest(_) => StatusCode::BAD_REQUEST, // 400
            ApiError::Forbidden(_) => StatusCode::FORBIDDEN, // 403
            ApiError::Internal(_) => StatusCode::INTERNAL_SERVER_ERROR, // 500
            ApiError::WebSocket(_) => StatusCode::INTERNAL_SERVER_ERROR, // 500
        };

        // Construimos el cuerpo JSON de la respuesta.
        // self.to_string() usa el mensaje definido en #[error("...")] de arriba.
        let body = Json(ErrorResponse {
            status: status.as_u16(),
            message: self.to_string(),
        });

        // (status, body) implementa IntoResponse en axum, asÃ­ que solo lo convertimos.
        (status, body).into_response()
    }
}

// Result "alias" para no escribir Result<T, ApiError> en cada funciÃ³n.
// Con esto en el resto del proyecto se puede escribir: fn algo() -> ApiResult<Medicion>
pub type ApiResult<T> = Result<T, ApiError>;

// Permite usar `?` directamente con consultas sqlx dentro de los handlers y servicios.
// El detalle del error se registra en el log, pero NO se envÃ­a al cliente.
impl From<sqlx::Error> for ApiError {
    fn from(error: sqlx::Error) -> Self {
        tracing::error!("Error de base de datos: {error}");
        ApiError::Internal("error de base de datos".to_string())
    }
}

/// Lee CORS_ORIGINS como una lista separada por comas.
///
/// Ejemplo:
/// CORS_ORIGINS=http://localhost:5500,http://127.0.0.1:5500
///
/// Sin configuraciÃ³n, no se autorizan orÃ­genes
/// de navegador adicionales.
fn configurar_cors() -> anyhow::Result<CorsLayer> {
    let origenes = env::var("CORS_ORIGINS").unwrap_or_default();

    let mut permitidos = Vec::new();

    for origen in origenes.split(',') {
        let origen = origen.trim();

        if origen.is_empty() {
            continue;
        }

        let header = HeaderValue::from_str(origen)
            .with_context(|| format!("Origen CORS invÃ¡lido: {origen}"))?;

        permitidos.push(header);
    }

    let cors = CorsLayer::new()
        .allow_methods([
            Method::GET,
            Method::POST,
            Method::PUT,
            Method::PATCH,
            Method::DELETE,
            Method::OPTIONS,
        ])
        .allow_headers([
            axum::http::header::CONTENT_TYPE,
            axum::http::header::AUTHORIZATION,
        ])
        .allow_origin(permitidos);

    Ok(cors)
}

async fn senal_de_apagado() {
    let ctrl_c = async {
        if let Err(error) = signal::ctrl_c().await {
            tracing::error!("No se pudo escuchar Ctrl+C: {error}");
            std::future::pending::<()>().await
        }
    };

    #[cfg(unix)]
    let terminar = async {
        match signal::unix::signal(signal::unix::SignalKind::terminate()) {
            Ok(mut senal) => {
                senal.recv().await;
            }
            Err(error) => {
                tracing::error!("No se pudo escuchar SIGTERM: {error}");
                std::future::pending::<()>().await
            }
        }
    };

    #[cfg(not(unix))]
    let terminar = std::future::pending::<()>();

    tokio::select! {
        _ = ctrl_c => {},
        _ = terminar => {},
    }

    tracing::info!("SeÃ±al de apagado recibida. Cerrando servidor...");
}

    /// Identificador del tipo de sensor.

    pub valor_maximo: f64,

// ============================================================
// VALIDACIÃ“N DE RANGOS
// ============================================================

impl RangoSensor {
    /// Comprueba si los lÃ­mites son vÃ¡lidos.
    pub fn limites_validos(&self) -> bool {
        self.valor_minimo.is_finite()
            && self.valor_maximo.is_finite()
            && self.valor_minimo < self.valor_maximo
    }

    /// Determina si un valor estÃ¡ fuera de los lÃ­mites.
    ///
    /// Devuelve:
    /// - Some("por_debajo") cuando es inferior al mÃ­nimo.
    /// - Some("por_encima") cuando supera el mÃ¡ximo.
    /// - None cuando estÃ¡ dentro del rango, el rango estÃ¡
    ///   desactivado o los valores no son vÃ¡lidos.
    pub fn evaluar_valor(&self, valor: f64) -> Option<&'static str> {
        if !self.activo || !self.limites_validos() || !valor.is_finite() {
            return None;
        }

        if valor < self.valor_minimo {
            Some("por_debajo")
        } else if valor > self.valor_maximo {
            Some("por_encima")
        } else {
            None
        }
    }
}

            valor_minimo,
            valor_maximo,
            medido_en
        FROM vw_alertas_detalle
        WHERE ($1::text IS NULL OR estado = $1)
          AND ($2::uuid IS NULL OR aula_id = $2)
        ORDER BY generada_en DESC
        LIMIT $3
        "#,

struct CambioEstado {

async fn cambiar_estado(
    Extension(pool): Extension<PgPool>,
    Path(id): Path<Uuid>,
    Json(datos): Json<CambioEstado>,
) -> ApiResult<Json<Value>> {
    let nuevo_estado = datos.estado.trim().to_lowercase();

    if !["revisada", "resuelta"].contains(&nuevo_estado.as_str()) {
        return Err(ApiError::BadRequest(
            "estado debe ser: revisada o resuelta".into(),
        ));
    }

    // Transiciones permitidas:
    //
    // activa   -> revisada
    // activa   -> resuelta
    // revisada -> resuelta
    //
    // Una alerta resuelta no puede reabrirse.

    let actualizada: Option<(Uuid,)> = sqlx::query_as(
        r#"
        UPDATE alertas
        SET
            estado = $2,
            resuelta_en = CASE
                WHEN $2 = 'resuelta' THEN NOW()
                ELSE NULL
            END
        WHERE id = $1
          AND (
              (estado = 'activa'
                  AND $2 IN ('revisada', 'resuelta'))
              OR
              (estado = 'revisada'
                  AND $2 = 'resuelta')
          )
        RETURNING id
        "#,
    )
    .bind(id)
    .bind(&nuevo_estado)
    .fetch_optional(&pool)
    .await?;

    if actualizada.is_some() {
        return Ok(Json(json!({
            "id": id,
            "estado": nuevo_estado,
            "mensaje": "Estado actualizado correctamente"
        })));
    }

    // Si no se actualizo, determinar si no existe
    // o si la transicion solicitada no esta permitida.

    let estado_actual: Option<(String,)> =
        sqlx::query_as("SELECT estado FROM alertas WHERE id = $1")
            .bind(id)
            .fetch_optional(&pool)
            .await?;

    match estado_actual {
        None => Err(ApiError::NotFound(format!("No existe la alerta {id}"))),

        Some((estado,)) => Err(ApiError::BadRequest(format!(
            "No se permite cambiar una alerta de '{estado}' a '{nuevo_estado}'"
        ))),
    }
}

        sqlx::query_as("SELECT codigo, nombre FROM aulas WHERE id = $1")

            .await?;

        aula.ok_or_else(|| ApiError::NotFound(format!("No existe el aula {aula_id}")))?;

    // Ãšltimo valor de cada variable.
    let filas: Vec<(String, String, f64, DateTime<Utc>)> = sqlx::query_as(

           FROM vw_mediciones_detalle
          WHERE aula_id = $1
          ORDER BY tipo_sensor, medido_en DESC",
    )
    .bind(aula_id)
    .fetch_all(&pool)
    .await?;

    let mut valores = Map::new();
    for (tipo, unidad, valor, medido_en) in filas {
        valores.insert(
            tipo,
            json!({ "valor": valor, "unidad": unidad, "medido_en": medido_en }),
        );
    }

    // Ãšltimo Ã­ndice de confort (0â€“100).
    let indice: Option<(f64, DateTime<Utc>)> = sqlx::query_as(
        "SELECT valor, calculado_en FROM indices_confort

    .bind(aula_id)
    .fetch_optional(&pool)
    .await?;

    let confort_json = indice.map(|(valor, calculado_en)| {
        json!({
            "valor": valor,
            "estado": confort::obtener_estado_confort(valor / 100.0),
            "calculado_en": calculado_en,
        })
    });

    let (alertas_activas,): (i64,) = sqlx::query_as(
        "SELECT COUNT(*) FROM vw_alertas_detalle WHERE aula_id = $1 AND estado = 'activa'",
    )
    .bind(aula_id)
    .fetch_one(&pool)
    .await?;

    Ok(Json(json!({
        "aula": { "id": aula_id, "codigo": codigo, "nombre": nombre },
        "valores": valores,
        "confort": confort_json,
        "alertas_activas": alertas_activas,
        "estado": if alertas_activas > 0 { "ALERTA" } else { "NORMAL" },
    })))
}

async fn crear_sensor(

    if payload.nombre.trim().is_empty() {

        }
    }

    let resultado = sqlx::query_as::<_, Sensor>(
        r#"
        INSERT INTO sensores (
            dispositivo_id,
            tipo_sensor_id,
            nombre,
            pin_gpio,
            estado
        )
        VALUES ($1, $2, $3, $4, 'activo')
        RETURNING
            id,
            dispositivo_id,
            tipo_sensor_id,
            nombre,
            pin_gpio,
            estado,
            creado_en
        "#,
    )
    .bind(payload.dispositivo_id)
    .bind(payload.tipo_sensor_id)
    .bind(payload.nombre.trim())
    .bind(payload.pin_gpio)
    .fetch_one(&pool)
    .await;

        Err(error) => error_bd(error),

        0.0
    } else if temperatura < 26.0 {
        (temperatura - 21.0) / 5.0
    } else if temperatura < 31.0 {
        (31.0 - temperatura) / 5.0
    } else {
        0.0
    };

    indice.clamp(0.0, 1.0)
}

/// Normaliza la humedad relativa entre 0 y 1.
///
/// El intervalo de referencia es 40â€“60 %.
fn calcular_indice_humedad(humedad: f64) -> f64 {
    let indice: f64 = if humedad <= 20.0 {
        0.0
    } else if humedad < 40.0 {
        (humedad - 20.0) / 20.0
    } else if humedad <= 60.0 {
        1.0
    } else if humedad < 90.0 {
        (90.0 - humedad) / 30.0
    } else {
        0.0
    };

    indice.clamp(0.0, 1.0)
}

/// Calcula el Ã­ndice visual a partir de la

pub fn calcular_indice_visual(datos: &DatosAmbientales) -> Result<f64, String> {

/// Calcula el Ã­ndice acÃºstico a partir del nivel
/// de ruido ambiental.
///
/// La fÃ³rmula conserva los intervalos definidos
/// originalmente por el proyecto.
///
/// Para interpretarlo como nivel sonoro en dBA,
/// el sistema de adquisiciÃ³n debe proporcionar
/// mediciones calibradas en esa unidad.
pub fn calcular_indice_acustico(datos: &DatosAmbientales) -> Result<f64, String> {
    datos.validar()?;

    let ruido = datos.ruido;

    let indice: f64 = if ruido <= 40.0 {
        1.0
    } else if ruido <= 50.0 {
        1.0 - ((ruido - 40.0) / 100.0)
    } else if ruido <= 60.0 {
        0.9 - ((ruido - 50.0) * 0.02)
    } else if ruido <= 65.0 {
        0.7 - ((ruido - 60.0) * 0.08)
    } else if ruido < 70.0 {
        0.3 - ((ruido - 65.0) * 0.06)
    } else {
        0.0
    };

    Ok(indice.clamp(0.0, 1.0))
}

/// Calcula el Ã­ndice general de confort ambiental.
///
/// Promedia los Ã­ndices tÃ©rmico, visual y acÃºstico
/// con la misma ponderaciÃ³n.
///
/// Resultado: escala de 0 a 1.
pub fn calcular_indice_confort(datos: &DatosAmbientales) -> Result<f64, String> {
    datos.validar()?;

    let indice_termico = calcular_indice_termico(datos)?;

    let indice_visual = calcular_indice_visual(datos)?;

    let indice_acustico = calcular_indice_acustico(datos)?;

    let indice_global: f64 = (indice_termico + indice_visual + indice_acustico) / 3.0;

    Ok(indice_global.clamp(0.0, 1.0))
}

/// Calcula el promedio de un conjunto de mediciones.
///
/// Devuelve None si no hay valores vÃ¡lidos.
pub fn calcular_promedio(mediciones: &[Medicion]) -> Option<f64> {
    let mut suma = 0.0;
    let mut cantidad = 0usize;

    for medicion in mediciones {
        if medicion.valor.is_finite() {
            suma += medicion.valor;
            cantidad += 1;
        }
    }

    if cantidad == 0 {
        return None;
    }

    let promedio = suma / cantidad as f64;

    if promedio.is_finite() {
        Some(promedio)
    } else {
        None
    }
}

/// Obtiene el valor mÃ¡ximo registrado.
pub fn calcular_maximo(mediciones: &[Medicion]) -> Option<f64> {
    mediciones
        .iter()
        .map(|medicion| medicion.valor)
        .filter(|valor| valor.is_finite())
        .reduce(f64::max)
}

/// Obtiene el valor mÃ­nimo registrado.
pub fn calcular_minimo(mediciones: &[Medicion]) -> Option<f64> {
    mediciones
        .iter()
        .map(|medicion| medicion.valor)
        .filter(|valor| valor.is_finite())
        .reduce(f64::min)
}

// Estado compartido de todo el sistema de WebSocket.
// `broadcast::Sender` permite enviar un mensaje UNA vez y que TODOS los
// clientes suscritos (todas las conexiones abiertas) lo reciban.
// Es justo lo que necesitamos para "actualizaciones de mediciones en tiempo real":
// cuando llega una mediciÃ³n nueva, se difunde a todos los que estÃ¡n viendo el dashboard.
#[derive(Clone)]
pub struct WsState {
    pub tx: broadcast::Sender<String>,
}

impl WsState {
    // Crea el estado inicial. `capacity` es cuÃ¡ntos mensajes se guardan en el
    // buffer interno: si un cliente es muy lento leyendo, despuÃ©s de ese lÃ­mite
    // empieza a perder mensajes viejos (no se acumulan infinitamente en memoria).
    pub fn new(capacity: usize) -> Arc<Self> {
        let (tx, _rx) = broadcast::channel(capacity);
        // El _rx (receptor inicial) se descarta: cada cliente que se conecte
        // crearÃ¡ su propio receptor con tx.subscribe() mÃ¡s abajo.
        Arc::new(Self { tx })
    }

    // FunciÃ³n pÃºblica que usa el resto del sistema (la ruta POST /api/mediciones)
    // para publicar una nueva mediciÃ³n a todos los clientes.
    pub fn broadcast_measurement(&self, payload: String) {
        // send() devuelve Err solo si NO hay ningÃºn cliente escuchando.
        // Eso no es un error grave para nosotros, asÃ­ que lo ignoramos con `let _ =`.
        let _ = self.tx.send(payload);
    }
}

// Handler HTTP normal que "upgradea" la conexiÃ³n a WebSocket.
// Usa `Extension` (igual que el pool y la config) porque el router no usa `with_state`.
pub async fn ws_handler(
    ws: WebSocketUpgrade,
    Extension(state): Extension<Arc<WsState>>,
) -> impl IntoResponse {
    ws.on_upgrade(move |socket| handle_socket(socket, state))
}

// LÃ³gica de UNA conexiÃ³n WebSocket individual (se ejecuta una vez por cada cliente conectado).
async fn handle_socket(socket: WebSocket, state: Arc<WsState>) {
    let (mut sender, mut receiver) = socket.split();

    // Cada cliente se suscribe de forma independiente al canal de broadcast.
    let mut rx = state.tx.subscribe();

    info!("Nuevo cliente WebSocket conectado");

    // TAREA 1: reenvÃ­a cada mediciÃ³n nueva a ESTE cliente.
    let mut send_task = tokio::spawn(async move {
        loop {
            match rx.recv().await {
                Ok(msg) => {
                    // Si falla el envÃ­o, el cliente ya se desconectÃ³.
                    if sender.send(Message::Text(msg)).await.is_err() {
                        break;
                    }
                }
                // Cliente lento: se perdieron mensajes viejos, pero la conexiÃ³n sigue.
                Err(RecvError::Lagged(omitidos)) => {
                    warn!("Cliente WebSocket lento: se omitieron {omitidos} mensajes");
                }
                Err(RecvError::Closed) => break,
            }
        }
    });

    // TAREA 2: escucha lo que el cliente envÃ­a (cierre, mensajes de control).
    let mut recv_task = tokio::spawn(async move {
        while let Some(Ok(msg)) = receiver.next().await {
            match msg {
                Message::Text(text) => {
                    info!("Mensaje recibido del cliente: {}", text);
                }
                Message::Close(_) => {
                    info!("El cliente cerrÃ³ la conexiÃ³n");
                    break;
                }
                _ => {} // Ignoramos pings/pongs/binarios por ahora
            }
        }
    });

    // Cuando una de las dos tareas termina, se cancela la otra.
    tokio::select! {
        _ = (&mut send_task) => recv_task.abort(),
        _ = (&mut recv_task) => send_task.abort(),
    }

    warn!("Cliente WebSocket desconectado");
}
